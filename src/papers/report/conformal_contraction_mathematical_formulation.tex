\section*{Closed-Loop System}

The real closed-loop system controlled by the learned policy is
\[
\dot x=f(x,u_{\mathrm{learned}}(x,E)).
\]
Add and subtract the nominal model-based controller term:
\[
\dot x
=f(x,u_{\mathrm{learned}}(x,E))
=\hat f(x,u_{\mathrm{mpc}}(x,E))
+f(x,u_{\mathrm{learned}}(x,E))-\hat f(x,u_{\mathrm{mpc}}(x,E)).
\]
For this verification, the candidate path is the learned-planner path and the nominal path is the reference path.

\section*{Contraction Metric Condition}

Assume a learned contraction metric \(\hat M(x,t)\succ0\) satisfying
\[
mI\preceq \hat M(x,t)\preceq \bar m I
\]
and approximately
\[
\dot{\hat M}
+\hat M\frac{\partial \hat f_{\mathrm{cl}}}{\partial x}
+\frac{\partial \hat f_{\mathrm{cl}}}{\partial x}\hat M
\preceq -2\alpha \hat M .
\]
Plug in the implemented diagonal metric bounds.
\[
1.0I\preceq
\begin{bmatrix}
1.0&0\\
0&1.0
\end{bmatrix}
\preceq 1.0I
\]
Final result.
\[
\boxed{\hat M=M=\operatorname{diag}(1.0,1.0)\text{ is positive definite and bounded.}}
\]

\section*{Riemannian Energy}

Define the Riemannian energy along the minimizing geodesic \(\gamma\) connecting \(x\) and \(x_n\) as
\[
\hat E(\gamma(\cdot,t))
=
\int_0^1
\delta\gamma(s,t)^\top
\hat M(\gamma(s,t))
\delta\gamma(s,t)\,ds,
\qquad
\delta\gamma(s,t)=\frac{\partial\gamma(s,t)}{\partial s}.
\]
For sampled 2D waypoint verification with a constant diagonal metric, use
\[
e_i=x_{c,i}-x_{n,i}
=
\begin{bmatrix}
\Delta x_i\\
\Delta y_i
\end{bmatrix},
\qquad
E_i=e_i^\top M e_i .
\]
Plug in \(M=\operatorname{diag}(1.0,1.0)\).
\[
E_i
=
\begin{bmatrix}
\Delta x_i&\Delta y_i
\end{bmatrix}
\begin{bmatrix}
1.0&0\\
0&1.0
\end{bmatrix}
\begin{bmatrix}
\Delta x_i\\
\Delta y_i
\end{bmatrix}
=\Delta x_i^2+\Delta y_i^2
\]
The maximum energy is computed over sampled candidate and nominal trajectory points:
\[
E_{\max}=\max_i E_i=\max_i e_i^\top M e_i.
\]
Plug in the computed verification value.
\[
E_{\max}=0.018738
\]
Final equivalent sampled Euclidean radius.
\[
r_{\max}
=\sqrt{\frac{E_{\max}}{m}}
=\sqrt{\frac{0.018738}{1.0}}
=0.136886\ \mathrm{m}
\]
\[
\boxed{E_{\max}\text{ is computed, not assumed.}}
\]

\section*{Dynamic Conformal Score}

Conformal inference gives the dynamic model-error event
\[
\mathbb P\left[
\left\|f(x,u_{\mathrm{learned}}(x,E))-\hat f(x,u_{\mathrm{mpc}}(x,E))\right\|
\le q_{1-\delta}^{\mathrm{dyn}},
\ \forall t\in[0,T]
\right]\ge 1-\delta^{\mathrm{dyn}},
\]
where
\[
q_{1-\delta}^{\mathrm{dyn}}
=
\left\lceil(1-\delta^{\mathrm{dyn}})(N+1)\right\rceil
\text{-th smallest of }
(S_1^{\mathrm{dyn}},\ldots,S_N^{\mathrm{dyn}})
\]
and
\[
S_i^{\mathrm{dyn}}
=
\sup_{t\in[0,T]}
\left\|f(x,u_{\mathrm{learned}}(x_i,E_i))
-\hat f(x,u_{\mathrm{mpc}}(x_i,E_i))\right\|.
\]
For sampled paths, compute
\[
S^{\mathrm{dyn}}=\max_i\|v_{c,i}-v_{n,i}\|_2.
\]
Plug in the values.
\[
S^{\mathrm{dyn}}=0.136886,\qquad q_{1-\delta}^{\mathrm{dyn}}=0.897275
\]
\[
0.136886\le0.897275
\]
Final result.
\[
\boxed{S^{\mathrm{dyn}}\le q_{1-\delta}^{\mathrm{dyn}}}

\section*{Contraction Residual Score}

\[

Plug in the values.

\[
0.067456\le2.898368

Final result.

\boxed{S^{\mathrm{con}}\le q_{1-\delta}^{\mathrm{con}}}

\section*{Young's Inequality Step}

\[

\qquad
\forall\epsilon>0.
\]

\[
\mathbb P\left[
\dot{\hat E}(\gamma(\cdot,t))
\le
-(2\alpha-\epsilon)\hat E(\gamma(\cdot,t))
+\left(q_{1-\delta}^{\mathrm{dyn}}\right)^2\bar m\epsilon^{-1}
+q_{1-\delta}^{\mathrm{con}},
\ \forall t\in[0,T]
\right]
\ge
1-\delta^{\mathrm{dyn}}-\delta^{\mathrm{con}}.
\]
Plug in the values in the additive disturbance term.
\[
\left(q_{1-\delta}^{\mathrm{dyn}}\right)^2\bar m\epsilon^{-1}
